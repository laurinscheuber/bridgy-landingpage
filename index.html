2. Automated Visual Tests

Generate tests that verify design consistency:

button.spec.ts TypeScript 
// Generated by Bridgy
import { test, expect } from '@playwright/test';

test.describe('Button Component', () => {
  test('matches Figma design specs', async ({ page }) => {
    await page.goto('/components/button');
    
    const button = page.locator('.btn-primary');
    
    // Verify design tokens
    await expect(button).toHaveCSS('background-color', 'rgb(99, 102, 241)');
    await expect(button).toHaveCSS('padding', '16px 32px');
    await expect(button).toHaveCSS('border-radius', '8px');
    
    // Visual regression test
    await expect(page).toHaveScreenshot('button-primary.png');
  });
});
